import { useEffect, useState, useCallback } from 'react'
import {
  Card, Button, Space, Input, Select, List, Tag, Empty, Spin, message,
  Tabs, Descriptions, Typography, Row, Col, Statistic, Popconfirm,
} from 'antd'
import {
  ReloadOutlined, SearchOutlined, DeleteOutlined,
  CopyOutlined, BugOutlined,
} from '@ant-design/icons'
import { errorLogApi, type ErrorLog } from '../../api'
import { usePersistentState } from '../../hooks/usePersistentState'

const { Text } = Typography

// 状态颜色映射
const STATUS_COLOR: Record<string, string> = {
  new: 'red',
  resolved: 'green',
  ignored: 'default',
}
const STATUS_LABEL: Record<string, string> = {
  new: '新建',
  resolved: '已解决',
  ignored: '已忽略',
}

export default function ErrorLogs() {
  const [loading, setLoading] = useState(false)
  const [errorLogs, setErrorLogs] = useState<ErrorLog[]>([])
  const [selectedId, setSelectedId] = useState<number | null>(null)
  const [detail, setDetail] = useState<ErrorLog | null>(null)
  const [detailLoading, setDetailLoading] = useState(false)
  const [statusCounts, setStatusCounts] = useState<Record<string, number>>({})

  // 筛选条件（持久化：刷新后保持上次筛选状态）
  const [filterStatus, setFilterStatus] = usePersistentState<string | undefined>(
    'xh.errorLogs.filterStatus', undefined,
    { validator: (v): v is string | undefined => v === undefined || typeof v === 'string' },
  )
  const [filterType, setFilterType] = usePersistentState<string>(
    'xh.errorLogs.filterType', '',
    { validator: (v): v is string => typeof v === 'string' },
  )
  const [aiFormat, setAiFormat] = useState<'json' | 'markdown'>('json')
  const [aiContent, setAiContent] = useState('')
  const [aiLoading, setAiLoading] = useState(false)

  const loadList = useCallback(() => {
    setLoading(true)
    errorLogApi
      .list({ status: filterStatus, error_type: filterType || undefined, limit: 100 })
      .then((res) => {
        setErrorLogs(res.items)
        setStatusCounts(res.status_counts || {})
      })
      .catch(() => message.error('加载错误日志失败'))
      .finally(() => setLoading(false))
  }, [filterStatus, filterType])

  const loadDetail = useCallback((id: number) => {
    setDetailLoading(true)
    errorLogApi
      .get(id)
      .then((data) => setDetail(data))
      .catch(() => message.error('加载详情失败'))
      .finally(() => setDetailLoading(false))
  }, [])

  const loadAiContext = useCallback(
    (id: number, format: 'json' | 'markdown') => {
      setAiLoading(true)
      errorLogApi
        .getAiContext(id, format)
        .then((content) => setAiContent(content))
        .catch(() => message.error('加载 AI 上下文失败'))
        .finally(() => setAiLoading(false))
    },
    [],
  )

  useEffect(() => {
    loadList()
  }, [loadList])

  // 选中错误日志时加载详情和 AI 上下文
  useEffect(() => {
    if (selectedId) {
      loadDetail(selectedId)
      loadAiContext(selectedId, aiFormat)
    }
  }, [selectedId])

  const handleStatusChange = (id: number, status: string) => {
    errorLogApi
      .updateStatus(id, status)
      .then(() => {
        message.success('状态已更新')
        loadList()
        if (detail?.id === id) {
          setDetail({ ...detail, status: status as ErrorLog['status'] })
        }
      })
      .catch(() => message.error('更新状态失败'))
  }

  const handleDelete = (id: number) => {
    errorLogApi
      .delete(id)
      .then(() => {
        message.success('已删除')
        if (selectedId === id) {
          setSelectedId(null)
          setDetail(null)
        }
        loadList()
      })
      .catch(() => message.error('删除失败'))
  }

  const handleCopyAi = () => {
    navigator.clipboard.writeText(aiContent).then(() => message.success('已复制到剪贴板'))
  }

  const handleAiFormatChange = (fmt: 'json' | 'markdown') => {
    setAiFormat(fmt)
    if (selectedId) loadAiContext(selectedId, fmt)
  }

  // 格式化时间
  const fmtTime = (ts: string) => {
    try {
      return new Date(ts).toLocaleString('zh-CN')
    } catch {
      return ts
    }
  }

  // Tab 内容构建
  const tabItems = detail
    ? [
        {
          key: 'overview',
          label: '概览',
          children: (
            <Descriptions column={1} bordered size="small">
              <Descriptions.Item label="错误类型">
                <Tag color="red">{detail.error_type}</Tag>
              </Descriptions.Item>
              <Descriptions.Item label="错误消息">{detail.error_message}</Descriptions.Item>
              <Descriptions.Item label="发生时间">{fmtTime(detail.timestamp)}</Descriptions.Item>
              <Descriptions.Item label="请求方法">{detail.request_method || 'N/A'}</Descriptions.Item>
              <Descriptions.Item label="请求路径">{detail.request_path || 'N/A'}</Descriptions.Item>
              <Descriptions.Item label="客户端 IP">{detail.client_ip || 'N/A'}</Descriptions.Item>
              <Descriptions.Item label="状态">
                <Select
                  size="small"
                  value={detail.status}
                  style={{ width: 120 }}
                  onChange={(v) => handleStatusChange(detail.id, v)}
                  options={[
                    { label: '新建', value: 'new' },
                    { label: '已解决', value: 'resolved' },
                    { label: '已忽略', value: 'ignored' },
                  ]}
                />
              </Descriptions.Item>
            </Descriptions>
          ),
        },
        {
          key: 'stack',
          label: '堆栈信息',
          children: (
            <pre
              style={{
                background: '#f5f5f5',
                padding: 12,
                borderRadius: 4,
                fontSize: 12,
                overflow: 'auto',
                maxHeight: 400,
                whiteSpace: 'pre-wrap',
                wordBreak: 'break-all',
              }}
            >
              {detail.stack_trace || '(无堆栈信息)'}
            </pre>
          ),
        },
        {
          key: 'request',
          label: '请求参数',
          children: (
            <pre
              style={{
                background: '#f5f5f5',
                padding: 12,
                borderRadius: 4,
                fontSize: 12,
                overflow: 'auto',
                maxHeight: 400,
                whiteSpace: 'pre-wrap',
                wordBreak: 'break-all',
              }}
            >
              {JSON.stringify(
                {
                  method: detail.request_method,
                  path: detail.request_path,
                  params: detail.request_params,
                  headers: detail.request_headers,
                  client_ip: detail.client_ip,
                  user_agent: detail.user_agent,
                },
                null,
                2,
              )}
            </pre>
          ),
        },
        {
          key: 'env',
          label: '服务器环境',
          children: (
            <pre
              style={{
                background: '#f5f5f5',
                padding: 12,
                borderRadius: 4,
                fontSize: 12,
                overflow: 'auto',
                maxHeight: 400,
                whiteSpace: 'pre-wrap',
              }}
            >
              {JSON.stringify(detail.server_env, null, 2)}
            </pre>
          ),
        },
        {
          key: 'ai',
          label: 'AI 上下文',
          children: (
            <div>
              <Space style={{ marginBottom: 8 }}>
                <Select
                  size="small"
                  value={aiFormat}
                  style={{ width: 120 }}
                  onChange={handleAiFormatChange}
                  options={[
                    { label: 'JSON', value: 'json' },
                    { label: 'Markdown', value: 'markdown' },
                  ]}
                />
                <Button size="small" icon={<CopyOutlined />} onClick={handleCopyAi}>
                  复制
                </Button>
              </Space>
              <Spin spinning={aiLoading}>
                <pre
                  style={{
                    background: '#f5f5f5',
                    padding: 12,
                    borderRadius: 4,
                    fontSize: 12,
                    overflow: 'auto',
                    maxHeight: 400,
                    whiteSpace: 'pre-wrap',
                    wordBreak: 'break-all',
                  }}
                >
                  {aiContent || '(无 AI 上下文)'}
                </pre>
              </Spin>
            </div>
          ),
        },
      ]
    : []

  return (
    <div className="page-container">
      <h2 style={{ marginBottom: 16 }}>
        <BugOutlined style={{ marginRight: 8 }} />
        后台错误日志
      </h2>

      {/* 统计卡片 */}
      <Row gutter={16} style={{ marginBottom: 16 }}>
        <Col span={6}>
          <Card>
            <Statistic
              title="新建"
              value={statusCounts.new || 0}
              valueStyle={{ color: '#ff4d4f' }}
            />
          </Card>
        </Col>
        <Col span={6}>
          <Card>
            <Statistic
              title="已解决"
              value={statusCounts.resolved || 0}
              valueStyle={{ color: '#52c41a' }}
            />
          </Card>
        </Col>
        <Col span={6}>
          <Card>
            <Statistic title="已忽略" value={statusCounts.ignored || 0} />
          </Card>
        </Col>
        <Col span={6}>
          <Card>
            <Space direction="vertical" style={{ width: '100%' }}>
              <Button icon={<ReloadOutlined />} onClick={loadList} loading={loading} block>
                刷新
              </Button>
              <Popconfirm
                title="清理已解决/已忽略的错误日志"
                description="清理 30 天前的记录"
                onConfirm={() => {
                  errorLogApi.cleanup(30).then((r) => {
                    message.success(`已清理 ${r.deleted} 条`)
                    loadList()
                  })
                }}
              >
                <Button icon={<DeleteOutlined />} block>
                  批量清理
                </Button>
              </Popconfirm>
            </Space>
          </Card>
        </Col>
      </Row>

      <Row gutter={16}>
        {/* 左侧：错误列表 */}
        <Col span={10}>
          <Card title="错误列表">
            <Space style={{ marginBottom: 16 }} wrap>
              <Select
                allowClear
                placeholder="状态"
                style={{ width: 120 }}
                value={filterStatus}
                onChange={(v) => setFilterStatus(v)}
                options={[
                  { label: '新建', value: 'new' },
                  { label: '已解决', value: 'resolved' },
                  { label: '已忽略', value: 'ignored' },
                ]}
              />
              <Input
                placeholder="错误类型"
                style={{ width: 150 }}
                value={filterType}
                onChange={(e) => setFilterType(e.target.value)}
                onPressEnter={loadList}
                allowClear
              />
              <Button type="primary" icon={<SearchOutlined />} onClick={loadList}>
                搜索
              </Button>
            </Space>
            <Spin spinning={loading}>
              {errorLogs.length === 0 ? (
                <Empty description="暂无错误日志" />
              ) : (
                <List
                  size="small"
                  bordered
                  style={{ maxHeight: 500, overflow: 'auto' }}
                  dataSource={errorLogs}
                  renderItem={(item) => (
                    <List.Item
                      onClick={() => setSelectedId(item.id)}
                      style={{
                        cursor: 'pointer',
                        background: selectedId === item.id ? '#e6f4ff' : undefined,
                      }}
                      actions={[
                        <Popconfirm
                          key="delete"
                          title="确认删除？"
                          onConfirm={(e) => {
                            e?.stopPropagation()
                            handleDelete(item.id)
                          }}
                        >
                          <Button
                            type="text"
                            size="small"
                            danger
                            icon={<DeleteOutlined />}
                            onClick={(e) => e.stopPropagation()}
                          />
                        </Popconfirm>,
                      ]}
                    >
                      <div style={{ width: '100%' }}>
                        <Space size="small" wrap>
                          <Tag color={STATUS_COLOR[item.status]}>
                            {STATUS_LABEL[item.status] || item.status}
                          </Tag>
                          <Tag color="red">{item.error_type}</Tag>
                          {item.request_path && (
                            <Text style={{ fontSize: 12, color: '#999' }}>
                              {item.request_method} {item.request_path}
                            </Text>
                          )}
                        </Space>
                        <div style={{ marginTop: 4, fontSize: 13 }}>
                          {item.error_message.length > 80
                            ? `${item.error_message.slice(0, 80)}...`
                            : item.error_message}
                        </div>
                        <div style={{ fontSize: 11, color: '#999', marginTop: 2 }}>
                          {fmtTime(item.timestamp)}
                        </div>
                      </div>
                    </List.Item>
                  )}
                />
              )}
            </Spin>
          </Card>
        </Col>

        {/* 右侧：详情面板 */}
        <Col span={14}>
          <Card title="错误详情">
            <Spin spinning={detailLoading}>
              {!detail ? (
                <Empty description="选择左侧错误日志查看详情" />
              ) : (
                <Tabs items={tabItems} />
              )}
            </Spin>
          </Card>
        </Col>
      </Row>
    </div>
  )
}
