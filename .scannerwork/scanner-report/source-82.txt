"""登录策略选择器（设计文档 §4.3）

核心问题：现有 5 种登录方式平铺，无智能选择，用户需手动判断用哪种。

解决方案：自动评估当前环境，按优先级选择最优登录路径。

决策树：
1. 系统浏览器已登录闲鱼？ → CDP 连接（零风险，指纹完全真实）
2. browser-data 有有效 Cookie？ → Playwright 有头 + 复用 user_data_dir
3. 系统浏览器 Cookie 可解密？ → 从系统浏览器导入
4. 用户可扫码？ → Playwright 有头扫码登录
5. 兜底 → 手动 Cookie 注入

注意：本模块仅负责"选择策略"，不负责执行登录。
执行由对应的 route handler 完成。
"""
from __future__ import annotations

import socket
from dataclasses import dataclass
from enum import Enum
from pathlib import Path
from typing import Any

from xianyu_hunter.infra.logger import get_logger
from xianyu_hunter.infra.yaml_config import get_config

logger = get_logger()


class LoginStrategy(str, Enum):
    """登录策略枚举（按优先级排序）"""
    CDP_CONNECT = "cdp_connect"           # 策略A: CDP 连接系统浏览器
    REUSE_USERDATA = "reuse_userdata"     # 策略B: 复用 browser-data Cookie
    BROWSER_IMPORT = "browser_import"     # 策略C: 从系统浏览器导入 Cookie
    QR_SCAN = "qr_scan"                   # 策略D: Playwright 有头扫码
    COOKIE_INJECT = "cookie_inject"       # 策略E: 手动 Cookie 注入（兜底）


@dataclass
class StrategyEvaluation:
    """策略评估结果"""
    recommended: LoginStrategy
    reason: str
    available_strategies: list[LoginStrategy]
    details: dict[str, Any]

    def to_dict(self) -> dict[str, Any]:
        return {
            "recommended": self.recommended.value,
            "reason": self.reason,
            "available": [s.value for s in self.available_strategies],
            "details": self.details,
        }


class LoginStrategySelector:
    """登录策略选择器

    评估当前环境，按优先级推荐最优登录策略。
    不执行登录，仅提供决策依据。
    """

    # CDP 默认端口
    CDP_PORT = 9222

    # 闲鱼关键 Cookie 名称
    KEY_COOKIES = {"_m_h5_tk", "_m_h5_tk_enc", "unb", "sgcookie", "cookie2", "lg2"}

    def __init__(self, cdp_target_port: int = CDP_PORT):
        # 重命名实例属性 cdp_target_port，避免与类常量 CDP_PORT 大小写混淆（S1845）
        self.cdp_target_port = cdp_target_port

    def evaluate(self) -> StrategyEvaluation:
        """评估环境并推荐登录策略

        按优先级依次检测，返回第一个可用的策略。
        所有检测结果记录在 details 中，供前端展示。
        """
        details: dict[str, Any] = {}
        available: list[LoginStrategy] = []

        # 策略A: CDP 连接系统浏览器
        cdp_available = self._check_cdp_available()
        details["cdp_available"] = cdp_available
        if cdp_available:
            available.append(LoginStrategy.CDP_CONNECT)
            return StrategyEvaluation(
                recommended=LoginStrategy.CDP_CONNECT,
                reason="检测到系统浏览器已开启 CDP 调试端口，指纹完全真实，零检测风险",
                available_strategies=available,
                details=details,
            )

        # 策略B: 复用 browser-data Cookie
        has_valid_cookies = self._check_user_data_cookies()
        details["has_valid_user_data_cookies"] = has_valid_cookies
        if has_valid_cookies:
            available.append(LoginStrategy.REUSE_USERDATA)
            return StrategyEvaluation(
                recommended=LoginStrategy.REUSE_USERDATA,
                reason="browser-data 中存在有效闲鱼 Cookie，可直接复用",
                available_strategies=available,
                details=details,
            )

        # 策略C: 从系统浏览器导入 Cookie
        browser_importable = self._check_browser_importable()
        details["browser_importable"] = browser_importable
        if browser_importable:
            available.append(LoginStrategy.BROWSER_IMPORT)
            return StrategyEvaluation(
                recommended=LoginStrategy.BROWSER_IMPORT,
                reason="检测到系统浏览器有闲鱼 Cookie，可自动导入",
                available_strategies=available,
                details=details,
            )

        # 策略D: 扫码登录（始终可用，需要用户在电脑旁）
        available.append(LoginStrategy.QR_SCAN)
        details["qr_scan_available"] = True

        # 策略E: 手动 Cookie 注入（始终可用）
        available.append(LoginStrategy.COOKIE_INJECT)
        details["cookie_inject_available"] = True

        return StrategyEvaluation(
            recommended=LoginStrategy.QR_SCAN,
            reason="未检测到已有登录态，推荐扫码登录（需用户在电脑旁）",
            available_strategies=available,
            details=details,
        )

    def _check_cdp_available(self) -> bool:
        """检测系统浏览器是否已开启 CDP 调试端口

        CDP 模式优先级最高的原因：
        - 使用用户真实浏览器，指纹零异常
        - 无需注入任何 stealth 脚本
        - Cookie 天然有效，与日常浏览共享
        """
        try:
            with socket.create_connection(("127.0.0.1", self.cdp_target_port), timeout=1.0):
                return True
        # S5713: ConnectionRefusedError 与 socket.timeout 均为 OSError 的子类，仅保留父类
        except OSError:
            return False

    def _check_user_data_cookies(self) -> bool:
        """检测 browser-data 中是否有有效的闲鱼 Cookie

        检查 JSON 文件（主数据源）和 SQLite（从数据源）。
        """
        # 1. 检查 JSON 文件
        json_path = Path("data") / "cookies.json"
        if json_path.exists():
            try:
                import json
                data = json.loads(json_path.read_text(encoding="utf-8"))
                cookies = data.get("cookies", [])
                names = {c.get("name", "") for c in cookies}
                if self.KEY_COOKIES & names:
                    return True
            except (json.JSONDecodeError, OSError):
                pass

        # 2. 检查 SQLite
        cfg = get_config()
        user_data_dir = Path(cfg.browser.user_data_dir)
        if not user_data_dir.is_absolute():
            user_data_dir = Path.cwd() / user_data_dir
        cookie_db = user_data_dir / "Default" / "Network" / "Cookies"
        if not cookie_db.exists():
            return False
        try:
            import sqlite3
            with sqlite3.connect(f"file:{cookie_db}?mode=ro", uri=True) as conn:
                rows = conn.execute(
                    "SELECT name FROM cookies WHERE host_key LIKE ? OR host_key LIKE ?",
                    ("%goofish%", "%taobao%"),
                ).fetchall()
                names = {n for (n,) in rows}
                return bool(self.KEY_COOKIES & names)
        except Exception:
            return False

    def _check_browser_importable(self) -> bool:
        """检测系统浏览器是否有可导入的闲鱼 Cookie

        仅检测 Cookie 文件是否存在，不实际解密
        （解密是耗时操作，仅在用户选择此策略时执行）。
        """
        import os

        if os.name != "nt":
            return False

        # Edge Cookie 路径
        edge_cookie_path = Path(os.environ.get("LOCALAPPDATA", "")) / \
            "Microsoft" / "Edge" / "User Data" / "Default" / "Network" / "Cookies"
        if edge_cookie_path.exists():
            return True

        # Chrome Cookie 路径
        chrome_cookie_path = Path(os.environ.get("LOCALAPPDATA", "")) / \
            "Google" / "Chrome" / "User Data" / "Default" / "Network" / "Cookies"
        if chrome_cookie_path.exists():
            return True

        return False


# ============== 单例 ==============

_selector: LoginStrategySelector | None = None


def get_strategy_selector() -> LoginStrategySelector:
    """获取策略选择器单例"""
    global _selector
    if _selector is None:
        _selector = LoginStrategySelector()
    return _selector
