def test_brand_like_seller_nick_region_masked_nick_moves_to_brand() -> None:
    display = {
        "title": "镁光ddr4 32G 3200 笔记本内存条",
        "price": 268.0,
        "seller_nick": "镁光数码",
        "region": "牧***蓉",
        "seller_credit": "",
        "is_sold": False,
    }

    corrected, field_map = normalize_display_fields(display)

    assert corrected["brand"] == "镁光数码"
    assert corrected["seller_nick"] == "牧***蓉"
    assert corrected["region"] == ""
    assert "brand" in field_map
    assert field_map["brand"]["label"] == "品牌"

def test_stale_display_brand_is_cleared_when_title_does_not_match() -> None:
    display = {
        "title": "#笔记本电脑配件 DDR4-3200-32G笔记本内存#笔记",
        "brand": "联想",
        "price": 650.0,
        "seller_nick": "我喜欢工作",

        "seller_credit": "",
        "is_sold": False,

def test_raw_brand_is_ignored_when_title_does_not_match() -> None:
    brand = extract_brand(
        {"brand": "联想"},
        title="#笔记本电脑配件 DDR4-3200-32G笔记本内存#笔记",
        seller_candidate="我喜欢工作",
    )

    assert brand == ""


def test_raw_brand_is_kept_when_title_alias_matches() -> None:
    brand = extract_brand(
        {"brand": "联想"},
        title="ThinkPad T14 笔记本内存",
        seller_candidate="我喜欢工作",
    )

    assert brand == "联想"

# === detail 流程 description 兜底测试 ===
def test_infer_brand_from_description_finds_brand_when_title_misses() -> None:
    """标题未命中品牌时，从描述前 200 字符继续推断

    场景：闲鱼卖家标题写"32GB DDR4 3200MHz 笔记本内存条"（无品牌），
    但描述开头写"品牌：联想 Lenovo 出品"，应能从描述中提取联想。
    """
    desc = "品牌：联想 Lenovo 出品\n原装拆机，保修三个月"
    assert infer_brand_from_title(desc[:200]) == "联想"


def test_infer_brand_from_description_takes_first_match() -> None:
    """描述中有多个品牌关键词时，返回别名表中靠前的首个命中（与标题推断行为一致）"""
    desc = "对比了佳能和尼康的微单，最后选了索尼"
    # _BRAND_ALIASES 中相机类顺序：佳能 > 尼康 > 富士 > ... > 索尼
    # 但索尼在手机类已存在（"索尼", ("索尼", "sony")），需确认实际顺序
    # 实际别名表顺序：佳能(相机第一) 在 索尼(手机后段，但已存在) 之前
    result = infer_brand_from_title(desc[:200])
    assert result in {"佳能", "尼康", "索尼"}


def test_infer_brand_from_long_description_uses_first_200_chars_only() -> None:
    """描述超过 200 字符时，仅前 200 字符参与推断

    防止长描述末尾的无关品牌名误命中（如"我之前用过苹果，现在换华硕"）。
    detail 流程只取 desc[:200]，本测试验证该切片行为。
    """
    long_desc = "普通描述" * 50 + "联想 ThinkPad"  # 联想在第 201 字符之后
    # detail 流程切片后，"联想" 不在前 200 字符内
    assert infer_brand_from_title(long_desc[:200]) == ""
