from __future__ import annotations

from datetime import datetime, timezone
from types import SimpleNamespace

from xianyu_hunter.infra.item_display_sync import merge_item_display_from_detail


def test_detail_merge_preserves_existing_brand_when_detail_brand_empty() -> None:
    """detail 流程的 extract_brand(raw=None) 仅做标题推断，空返回不代表"权威判定无品牌"。

    修复前：merged["brand"] = detail.brand or "" 会清空旧 brand，导致用户点击标题
    触发 refresh 后品牌字段消失。修复后：brand 与 title/price 一致采用 set_if_present，
    仅当 detail 采集到非空品牌时才覆盖。
    """
    existing = {
        "title": "#笔记本电脑配件 DDR4-3200-32G笔记本内存#笔记",
        "brand": "联想",
        "seller_credit": "信用极好",
    }
    detail = SimpleNamespace(
        title="#笔记本电脑配件 DDR4-3200-32G笔记本内存#笔记",
        price=650.0,
        seller_id="209478264",
        region="广州",
        thumb_url="//img.alicdn.com/example.jpg",
        want_cnt=4,
        view_cnt=106,
        publish_time=datetime(2026, 6, 27, 12, 4, tzinfo=timezone.utc),
        is_sold=False,
        brand="",
    )

    merged = merge_item_display_from_detail(existing, detail)

    # 旧 brand 应被保留，不被空值清空
    assert merged["brand"] == "联想"
    assert merged["seller_credit"] == "信用极好"
    assert merged["price"] == 650.0
    assert merged["publish_time"] == "2026-06-27T12:04:00+00:00"


def test_detail_merge_overwrites_brand_when_detail_has_brand() -> None:
    """detail 采集到非空品牌时（标题命中别名表），应覆盖旧 brand。"""
    existing = {
        "title": "旧标题",
        "brand": "旧品牌",
    }
    detail = SimpleNamespace(
        title="ThinkPad X1 Carbon 笔记本电脑",
        price=8999.0,
        seller_id="seller_001",
        region="北京",
        thumb_url="//img.alicdn.com/new.jpg",
        want_cnt=10,
        view_cnt=200,
        publish_time=None,
        is_sold=False,
        brand="联想",
    )

    merged = merge_item_display_from_detail(existing, detail)

    assert merged["brand"] == "联想"
    assert merged["title"] == "ThinkPad X1 Carbon 笔记本电脑"


def test_detail_merge_preserves_existing_brand_when_no_existing_and_detail_empty() -> None:
    """无旧 brand 且 detail 也无品牌时，brand 字段不出现（与 set_if_present 一致）。"""
    existing = {"title": "无品牌商品"}
    detail = SimpleNamespace(
        title="无品牌商品",
        price=100.0,
        seller_id="seller_002",
        region="上海",
        thumb_url="//img.alicdn.com/no.jpg",
        want_cnt=0,
        view_cnt=10,
        publish_time=None,
        is_sold=False,
        brand="",
    )

    merged = merge_item_display_from_detail(existing, detail)

    # 无旧 brand + detail 无 brand → 不写入 brand 键
    assert "brand" not in merged or merged.get("brand") in (None, "")
