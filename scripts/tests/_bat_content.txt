@echo off
chcp 936 >nul 2>&1
REM 从 scripts/node-config.json 读取 Node.js 路径（配置化，避免硬编码）
REM 为什么用 PowerShell 读 JSON：cmd 难以解析 JSON，PowerShell 原生支持 ConvertFrom-Json
for /f "delims=" %%i in ('powershell -NoProfile -Command "$c = Get-Content '%~dp0node-config.json' -Raw -Encoding UTF8 | ConvertFrom-Json; $exe = $null; foreach ($p in $c.node.search_paths) { $e = [System.Environment]::ExpandEnvironmentVariables($p); if (Test-Path $e) { $exe = $e; break } }; if (-not $exe) { $cmd = Get-Command node -ErrorAction SilentlyContinue; if ($cmd) { $exe = $cmd.Source } }; Write-Output $exe"') do set "NODE_EXE=%%i"

if "%NODE_EXE%"=="" (
    echo [ERROR] 未找到 Node.js！
    echo 请配置 scripts\node-config.json 或将 node 加入 PATH
    pause
    exit /b 1
)

REM 把配置的 node 目录前置到 PATH，确保后续 npm/npx 调用都走指定版本
for %%d in ("%NODE_EXE%") do set "NODE_DIR=%%~fdp"
set "PATH=%NODE_DIR%;%PATH%"

echo ============================================
echo   闲鱼猎人 - 前端构建工具
echo ============================================
echo.
echo Node 路径: %NODE_EXE%
"%NODE_EXE%" --version
echo.

REM 脚本位于 scripts/ 子目录，切换到 frontend 目录
cd /d "%~dp0..\frontend"

echo [1/2] 正在清理旧构建产物...
if exist "..\src\xianyu_hunter\web\static\spa" (
    rmdir /s /q "..\src\xianyu_hunter\web\static\spa"
    echo   旧产物已清理
)

echo.
echo [2/2] 正在构建前端...
REM 直接用配置的 node.exe 调用 vite.js，不依赖 PATH 中的 node
"%NODE_EXE%" "node_modules\vite\bin\vite.js" build

if %ERRORLEVEL% NEQ 0 (
    echo.
    echo [ERROR] 构建失败，请查看上方错误信息
    pause
    exit /b 1
)

echo.
echo [3/3] 正在生成构建信息（版本号 + git sha）...
cd /d "%~dp0.."
python scripts\build_info.py
if errorlevel 1 (
    echo [WARN] build_info.py 执行失败，前端页面将显示 "unknown"
)

echo.
echo ============================================
echo   构建完成，请重启服务并按 Ctrl+F5 强制刷新
echo ============================================
pause
